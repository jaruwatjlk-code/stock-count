function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "Stock Count";
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      var h = ["วันที่นับ","เวลาเริ่ม","แผนก","เวลาเสร็จ",
               "ระยะเวลา (นาที)","ผู้บันทึก","หมายเหตุ","บันทึกเมื่อ"];
      sheet.appendRow(h);
      sheet.getRange(1,1,1,h.length)
           .setFontWeight("bold")
           .setBackground("#1a365d")
           .setFontColor("#ffffff");
      sheet.setFrozenRows(1);
    }
    var p = e.parameter;
    sheet.appendRow([
      p.date     || "",
      p.startTime|| "",
      p.dept     || "",
      p.finishTime|| "",
      parseInt(p.duration)||0,
      p.recorder || "",
      p.remarks  || "",
      Utilities.formatDate(new Date(),
        Session.getScriptTimeZone(),"dd/MM/yyyy HH:mm")
    ]);
    sheet.autoResizeColumns(1, 8);
    return ContentService
      .createTextOutput("OK")
      .setMimeType(ContentService.MimeType.TEXT);
  } catch(err) {
    return ContentService
      .createTextOutput("ERROR:" + err.toString())
      .setMimeType(ContentService.MimeType.TEXT);
  }
}
function doGet() {
  return ContentService
    .createTextOutput("Stock Count API OK")
    .setMimeType(ContentService.MimeType.TEXT);
}
